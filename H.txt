using UnityEngine;
using StarterAssets; 

public class JoystickToPlayer : MonoBehaviour
{
    [Header("Joystick Input (Hanya untuk Mobile)")]
    public FixedJoystick joystick;

    private StarterAssetsInputs playerInputs;

    void Start()
    {
        // Mengambil komponen StarterAssetsInputs di karakter ini
        playerInputs = GetComponent<StarterAssetsInputs>();
        
        // --- PENGECEKAN PLATFORM ---
#if UNITY_EDITOR || UNITY_STANDALONE
        // Jika main di PC, sembunyikan UI Joystick agar layar bersih
        if (joystick != null && joystick.transform.parent != null)
        {
            joystick.transform.parent.gameObject.SetActive(false);
        }
#endif
    }

#if UNITY_ANDROID || UNITY_IOS
    // Update ini HANYA akan berjalan jika di-build ke Android/iOS
    void Update()
    {
        if (joystick != null && playerInputs != null)
        {
            // Baca nilai joystick dan timpa nilai input dari Starter Assets
            Vector2 input = new Vector2(joystick.Horizontal, joystick.Vertical);
            playerInputs.MoveInput(input);
        }
    }
#endif
}
