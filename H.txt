using UnityEngine;
using StarterAssets; // Wajib ada agar bisa akses script StarterAssetsInputs

public class JoystickToPlayer : MonoBehaviour
{
    [Header("Hubungkan Joystick Disini")]
    public FixedJoystick joystick;

    // Variabel untuk menyimpan komponen StarterAssetsInputs di player ini
    private StarterAssetsInputs playerInputs;

    void Start()
    {
        // Otomatis mencari komponen StarterAssetsInputs di objek ini (Player)
        playerInputs = GetComponent<StarterAssetsInputs>();
        
        if (playerInputs == null)
        {
            Debug.LogError("Script StarterAssetsInputs tidak ditemukan di objek ini!");
        }
    }

    void Update()
    {
        // Jika joystick sudah dimasukkan dan komponen player ada
        if (joystick != null && playerInputs != null)
        {
            // Baca nilai joystick dan kirim ke pergerakan player
            Vector2 input = new Vector2(joystick.Horizontal, joystick.Vertical);
            playerInputs.MoveInput(input);
        }
    }
}
